{{- /* Numbered pagination. Overrides the theme's prev/next-only partial. Layout: < 上一页 1 … 4 5 [6] 7 8 … 20 下一页 > - First and last page are always shown; two pages on each side of the current one are shown; gaps collapse to an ellipsis. A gap of exactly one page is shown as the page itself, so "1 … 3" never appears. - The current page is plain text with aria-current="page". - Nothing is rendered when there is only one page. */ -}} {{- $p := .Paginator -}} {{- if gt $p.TotalPages 1 -}} {{- $cur := $p.PageNumber -}} {{- $total := $p.TotalPages -}} {{- $gap := false -}}
{{- if $p.HasPrev }} < 上一页 {{- else }} < 上一页 {{- end }}
    {{- range $p.Pagers -}} {{- $n := .PageNumber -}} {{- $near := le (math.Abs (sub $n $cur)) 2 -}} {{- $edge := or (eq $n 1) (eq $n $total) -}} {{- $fill := or (and (eq $n 2) (le $cur 5)) (and (eq $n (sub $total 1)) (ge $cur (sub $total 4))) -}} {{- if or $near $edge $fill -}} {{- $gap = false -}} {{- if eq $n $cur }}
  1. {{ $n }}
    2. {{- else }}
  2. {{ $n }}
    3. {{- end -}} {{- else if not $gap -}} {{- $gap = true }}
  3. …
    4. {{- end -}} {{- end }}
{{- if $p.HasNext }} 下一页 > {{- else }} 下一页 > {{- end }}

第 {{ $cur }} / {{ $total }} 页

{{- end -}}
